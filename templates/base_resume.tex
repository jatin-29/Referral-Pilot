\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\tabcolsep}{0pt}

\IfFileExists{enumitem.sty}{%
  \usepackage{enumitem}%
  \newenvironment{rpitems}%
    {\begin{itemize}[leftmargin=1.3em,itemsep=0.5pt,topsep=1pt,parsep=0pt,partopsep=0pt]\small}%
    {\end{itemize}}%
}{%
  \newenvironment{rpitems}%
    {\vspace{-5pt}\begin{itemize}\setlength{\itemsep}{0.5pt}\setlength{\parskip}{0pt}\small}%
    {\end{itemize}\vspace{-4pt}}%
}

\newcommand{\rpsection}[1]{%
  \vspace{6pt}{\color{rpaccent}\large\bfseries\MakeUppercase{#1}}\par\vspace{-2pt}%
  {\color{rpaccent}\rule{\linewidth}{0.6pt}}\par\vspace{2pt}%
}
\newcommand{\rpheading}[2]{\textbf{#1}\hfill{\small #2}\par}

%# ---------------------------------------------------------------- sections ---
%% macro summary()
\rpsection{Summary}
{\small \VAR{doc.summary|rich}}\par
%% endmacro

%% macro education()
\rpsection{Education}
%% for edu in doc.education
\rpheading{\VAR{edu.institution|tex}}{\VAR{edu.dates|tex}}
{\small\textit{\VAR{edu.degree|tex}}%
%% if edu.score
\ \textbar{} \VAR{edu.score|tex}%
%% endif

%% if edu.coursework
{\small Relevant coursework: \VAR{edu.coursework|items}}\par
%% endif
\vspace{2pt}
%% endfor
%% endmacro

%% macro skills()
\rpsection{Technical Skills}
{\small
\begin{tabularx}{\linewidth}{@{}>{\bfseries}p{0.17\linewidth}>{\raggedright\arraybackslash}X@{}}
%% for category, values in doc.skills
\VAR{category|tex} & \VAR{values|items} \\[1pt]
%% endfor
\end{tabularx}}
%% endmacro

%% macro experience()
\rpsection{Experience}
%% for job in doc.experience
\rpheading{\VAR{job.role|tex} \textnormal{\textbar{} \VAR{job.company|tex}%
%% if job.location

%% endif
}}{\VAR{job.dates|tex}}
\begin{rpitems}
%% for bullet in job.bullets
  \item \VAR{bullet|rich}
%% endfor
\end{rpitems}
\vspace{2pt}
%% endfor
%% endmacro

%% macro projects()
\rpsection{Projects}
%% for project in doc.projects
\textbf{\VAR{project.name|tex}} {\small\textbar{} \VAR{project.tech|items}}%
%% if project.link
\hfill{\small\href{\VAR{project.link|url}}{\VAR{project.link_label|tex}}}%
%% endif
\par
\begin{rpitems}
%% for bullet in project.bullets
  \item \VAR{bullet|rich}
%% endfor
\end{rpitems}
\vspace{2pt}
%% endfor
%% endmacro

%% macro achievements()
\rpsection{Achievements}
\begin{rpitems}
%% for item in doc.achievements
  \item \VAR{item|rich}
%% endfor
\end{rpitems}
%% endmacro
